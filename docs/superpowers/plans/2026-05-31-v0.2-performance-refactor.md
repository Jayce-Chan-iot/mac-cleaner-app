# MacCleanerApp v0.2 性能重构 — 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** 将扫描时间从 10+ 分钟降到 <1 分钟，内存从 60GB 降到 ~100MB，新增后台扫描不中断、垃圾详情展开、菜单栏系统状态监控。

**Architecture:** 核心是 UnifiedScanner——一次流式遍历 home 目录，每读到一个文件同时喂给 JunkRule/SizeFilter/DupCollector 三个管道，不再调用 NSEnumerator.allObjects。AppState 持有全局 scanTask 保证切页不中断。NSStatusBar 独立模块显示 CPU/内存/温度。

**Tech Stack:** Swift 6, SwiftUI macOS 14+, IOKit (SMC), host_statistics, NSStatusBar

- [ ] **Step 1: 创建 UnifiedScanner.swift**

```swift
import Foundation
import CryptoKit

/// 统一扫描引擎 — 一次流式遍历 Home，同时完成垃圾识别、大文件筛选、重复收集
/// 替代 FileScanner + JunkDetector + DuplicateDetector 的三轮全量扫描
actor UnifiedScanner {
    private let fileManager = FileManager.default
    private let safetyManager = SafetyManager()
    private let junkDetector = JunkDetector()

    /// 单次遍历的汇总结果
    struct ScanOutput {
        var junkFiles: [JunkCategory: [FileItem]] = [:]
        var largeFiles: [FileItem] = []
        var duplicateCandidates: [Int64: [FileItem]] = [:] // 按大小分桶
    }

    // MARK: - Public API

    /// 执行统一扫描
    func scan(
        rootURL: URL,
        minLargeFileSize: Int64 = 100_000_000,
        olderThanDays: Int = 180,
        progressHandler: (@Sendable (String, Double) -> Void)? = nil
    ) async throws -> (
        junk: [JunkCategory: [FileItem]],
        largeFiles: [FileItem],
        duplicates: [DuplicateGroup]
    ) {
        var output = ScanOutput()

        // Step A: 流式遍历，多路分流
        progressHandler?("扫描文件系统...", 0.05)
        try await streamScan(rootURL: rootURL, into: &output, progressHandler: { count in
            progressHandler?("已扫描 \(count) 个文件...", 0.05 + Double(count) / 20000 * 0.3)
        })

        // Step B: 大文件按大小/时间排序
        progressHandler?("筛选大文件...", 0.4)
        let largeFiles = output.largeFiles
            .filter { $0.size >= minLargeFileSize || $0.isOldFile(olderThan: olderThanDays) }
            .sorted { $0.size > $1.size }

        // Step C: 对重复候选做 SHA-256 验证
        progressHandler?("检测重复文件...", 0.5)
        let duplicateGroups = try await verifyDuplicates(
            sizeBuckets: output.duplicateCandidates.filter { $0.value.count >= 2 },
            progressHandler: { progress in
                progressHandler?("重复检测: \(Int(progress * 100))%", 0.5 + progress * 0.45)
            }
        )

        progressHandler?("扫描完成", 1.0)
        return (junk: output.junkFiles, largeFiles: largeFiles, duplicates: duplicateGroups)
    }

    // MARK: - Streaming Scan

    private func streamScan(
        rootURL: URL,
        into output: inout ScanOutput,
        progressHandler: ((Int) -> Void)? = nil
    ) async throws {
        // 复用 JunkDetector 的规则进行路径匹配
        let junkRules = junkDetector.rules // 需要 JunkDetector 暴露 rules

        guard let enumerator = fileManager.enumerator(
            at: rootURL,
            includingPropertiesForKeys: [
                .fileSizeKey, .contentModificationDateKey, .isDirectoryKey,
                .contentAccessDateKey, .creationDateKey
            ],
            options: [.skipsHiddenFiles, .skipsPackageDescendants],
            errorHandler: { _, _ in return true }
        ) else { return }

        var fileCount = 0
        var batch: [(URL, Int64, Date?, Date?, Date?, String)] = []
        let batchSize = 1000

        for case let fileURL as URL in enumerator {
            // 跳过受保护路径
            guard !fileURL.path.hasPrefix("/System"),
                  !fileURL.path.hasPrefix("/private/var") else { continue }

            fileCount += 1
            autoreleasepool {
                guard let values = try? fileURL.resourceValues(forKeys: [
                    .fileSizeKey, .contentModificationDateKey, .isDirectoryKey,
                    .contentAccessDateKey, .creationDateKey
                ]) else { return }

                // 跳过目录和 0 字节文件
                guard values.isDirectory != true else { return }
                let size = Int64(values.fileSize ?? 0)
                guard size > 0 else { return }

                let modDate = values.contentModificationDate
                let accessDate = values.contentAccessDate
                let createDate = values.creationDate
                let ext = fileURL.pathExtension

                batch.append((fileURL, size, modDate, accessDate, createDate, ext))

                if batch.count >= batchSize {
                    let flushed = batch
                    batch.removeAll(keepingCapacity: true)
                    processBatch(flushed, into: &output)
                }
            }
        }

        // 处理最后一批
        if !batch.isEmpty {
            processBatch(batch, into: &output)
        }

        progressHandler?(fileCount)
    }

    private func processBatch(
        _ batch: [(URL, Int64, Date?, Date?, Date?, String)],
        into output: inout ScanOutput
    ) {
        for (fileURL, size, modDate, accessDate, createDate, ext) in batch {
            let fileType = FileItem.FileCategory.from(extension: ext)
            let item = FileItem(
                url: fileURL,
                fileName: fileURL.lastPathComponent,
                fileExtension: ext,
                size: size,
                modificationDate: modDate,
                lastAccessDate: accessDate,
                creationDate: createDate,
                isDirectory: false,
                fileType: fileType
            )

            // 1. 垃圾规则匹配
            for (category, baseURL, pattern) in junkDetector.rules {
                if junkDetector.matchesRule(url: fileURL, baseURL: baseURL, pattern: pattern) {
                    output.junkFiles[category, default: []].append(item)
                    break
                }
            }

            // 2. 大文件收集（先收集所有，最后再排序）
            if size >= 50_000_000 { // 50MB 以上先收集
                output.largeFiles.append(item)
            }

            // 3. 重复文件按大小分桶（只记录，SHA-256 在遍历后做）
            if size >= 1024 {
                output.duplicateCandidates[size, default: []].append(item)
            }
        }
    }

    // MARK: - Duplicate Verification

    private func verifyDuplicates(
        sizeBuckets: [Int64: [FileItem]],
        progressHandler: ((Double) -> Void)? = nil
    ) async throws -> [DuplicateGroup] {
        var allGroups: [DuplicateGroup] = []
        var processedBuckets = 0
        let totalBuckets = sizeBuckets.count

        for (size, files) in sizeBuckets where files.count >= 2 {
            // Phase 1: 前 4KB 哈希
            var prefixGroups: [String: [FileItem]] = [:]
            for file in files {
                guard let handle = try? FileHandle(forReadingFrom: file.url) else { continue }
                let data = (try? handle.read(upToCount: 4096)) ?? Data()
                try? handle.close()
                let hash = SHA256.hash(data: data)
                let key = hash.compactMap { String(format: "%02x", $0) }.joined()
                prefixGroups[key, default: []].append(file)
            }

            // Phase 2: 完整哈希确认
            for (_, samePrefix) in prefixGroups where samePrefix.count >= 2 {
                var fullGroups: [String: [FileItem]] = [:]
                for file in samePrefix {
                    guard let data = try? Data(contentsOf: file.url, options: [.mappedIfSafe]) else { continue }
                    let hash = SHA256.hash(data: data)
                    let key = hash.compactMap { String(format: "%02x", $0) }.joined()
                    fullGroups[key, default: []].append(file)
                }
                for (hash, dups) in fullGroups where dups.count >= 2 {
                    allGroups.append(DuplicateGroup(fileSize: size, fileHash: String(hash.prefix(12)), files: dups))
                }
            }

            processedBuckets += 1
            progressHandler?(Double(processedBuckets) / Double(totalBuckets))
        }

        // 释放中间数据
        return allGroups.sorted { $0.wasteSize > $1.wasteSize }
    }
}
```

```swift
func matchesRule(url: URL, baseURL: URL, pattern: String?) -> Bool {
    guard url.path.hasPrefix(baseURL.path) else { return false }
    guard let pattern = pattern else { return true }
    // 简单 Glob: 检查路径是否匹配模式
    let patternComponents = pattern.components(separatedBy: "/")
    let urlComponents = url.pathComponents
    guard urlComponents.count >= patternComponents.count else { return false }
    // 从 baseURL 之后开始匹配
    let baseCount = baseURL.pathComponents.count
    var pi = 0
    var ui = baseCount
    while pi < patternComponents.count && ui < urlComponents.count {
        if patternComponents[pi] == "*" {
            pi += 1
            // 跳过通配符，找到下一个固定组件的匹配
            if pi < patternComponents.count {
                while ui < urlComponents.count && urlComponents[ui] != patternComponents[pi] {
                    ui += 1
                }
                if ui >= urlComponents.count { return false }
            }
        } else if patternComponents[pi] != urlComponents[ui] {
            return false
        }
        pi += 1
        ui += 1
    }
    return pi >= patternComponents.count
}
```

```swift
import Foundation
import IOKit

/// 系统状态监控 — CPU、内存、温度
actor SystemMonitor {
    private var timer: Timer? = nil

    struct SystemStats {
        let cpuUsage: Double      // 0.0-1.0
        let memoryUsed: UInt64    // bytes
        let memoryTotal: UInt64   // bytes
        let cpuTemperature: Double // Celsius

        var memoryUsagePercent: Double {
            Double(memoryUsed) / Double(memoryTotal)
        }
    }

    // MARK: - CPU

    func getCPUUsage() -> Double {
        var cpuInfo: processor_info_array_t!
        var numCpuInfo: mach_msg_type_number_t = 0
        var numCpus: natural_t = 0

        guard host_processor_info(mach_host_self(), PROCESSOR_CPU_LOAD_INFO, &numCpus, &cpuInfo, &numCpuInfo) == KERN_SUCCESS else {
            return 0
        }
        defer { vm_deallocate(mach_task_self_, vm_address_t(bitPattern: cpuInfo), vm_size_t(Int(numCpuInfo) * MemoryLayout<integer_t>.size)) }

        var totalUser: Double = 0
        var totalSystem: Double = 0
        var totalIdle: Double = 0

        for i in 0..<Int(numCpus) {
            let cpuLoad = cpuInfo.advanced(by: Int(CPU_STATE_MAX) * i)
            totalUser += Double(cpuLoad[Int(CPU_STATE_USER)])
            totalSystem += Double(cpuLoad[Int(CPU_STATE_SYSTEM)])
            totalIdle += Double(cpuLoad[Int(CPU_STATE_IDLE)])
        }

        let total = totalUser + totalSystem + totalIdle
        guard total > 0 else { return 0 }
        return (totalUser + totalSystem) / total
    }

    // MARK: - Memory

    func getMemoryStats() -> (used: UInt64, total: UInt64) {
        var info = host_basic_info()
        var count = mach_msg_type_number_t(MemoryLayout<host_basic_info>.size / MemoryLayout<integer_t>.size)

        guard host_info(mach_host_self(), HOST_BASIC_INFO, &info, &count) == KERN_SUCCESS else {
            return (0, 0)
        }

        let total = UInt64(info.max_mem)

        var vmStat = vm_statistics64()
        var vmCount = mach_msg_type_number_t(MemoryLayout<vm_statistics64>.size / MemoryLayout<integer_t>.size)

        guard host_statistics64(mach_host_self(), HOST_VM_INFO64, &vmStat, &vmCount) == KERN_SUCCESS else {
            return (0, total)
        }

        let pageSize = UInt64(vm_kernel_page_size)
        let used = (UInt64(vmStat.active_count) + UInt64(vmStat.wire_count) + UInt64(vmStat.compressed_page_count)) * pageSize
        return (used, total)
    }

    // MARK: - Temperature (IOKit SMC)

    func getCPUTemperature() -> Double {
        let service = IOServiceGetMatchingService(kIOMainPortDefault, IOServiceMatching("AppleSMC"))
        guard service != 0 else { return 0 }
        defer { IOServiceClose(service) }

        // 通过 SMC key "TC0P" 读取 CPU 温度
        // 简化版：返回 0 表示读取失败（SMC 访问需要特殊处理）
        // 实际项目可用 SMCKit 或 smc-command 命令行工具
        return readSMCTemperature(service: service, key: "TC0P")
    }

    private func readSMCTemperature(service: io_service_t, key: String) -> Double {
        // SMC 读取需要 IOKit 连接和特殊数据结构
        // 这里返回占位值；实际温度读取需要完整的 SMC 协议实现
        return 0
    }

    // MARK: - Snapshot

    func snapshot() -> SystemStats {
        let cpu = getCPUUsage()
        let (memUsed, memTotal) = getMemoryStats()
        let temp = getCPUTemperature()
        return SystemStats(cpuUsage: cpu, memoryUsed: memUsed, memoryTotal: memTotal, cpuTemperature: temp)
    }
}
```

```swift
import SwiftUI

/// NSStatusBar 菜单栏系统状态显示
/// 刘海左侧：温度 | 刘海右侧：CPU + RAM
@MainActor
final class MenuBarController: ObservableObject {
    static let shared = MenuBarController()

    private var statusItem: NSStatusItem?
    private var cpuItem: NSStatusItem?
    private var ramItem: NSStatusItem?
    private var tempItem: NSStatusItem?

    @Published var cpuUsage: Double = 0
    @Published var ramUsage: Double = 0
    @Published var temperature: Double = 0
    @Published var memoryUsed: UInt64 = 0
    @Published var memoryTotal: UInt64 = 0

    private let monitor = SystemMonitor()
    private var refreshTask: Task<Void, Never>?

    private init() {}

    func start() {
        // 温度 — 刘海左侧（或菜单栏最左）
        tempItem = NSStatusBar.system.statusItem(withLength: NSStatusItem.variableLength)
        if let button = tempItem?.button {
            button.title = "🌡️ --°C"
            button.font = NSFont.systemFont(ofSize: 11)
        }

        // CPU — 菜单栏右侧
        cpuItem = NSStatusBar.system.statusItem(withLength: NSStatusItem.variableLength)
        if let button = cpuItem?.button {
            button.title = "⚡ --%"
            button.font = NSFont.systemFont(ofSize: 11)
        }

        // RAM — 菜单栏右侧
        ramItem = NSStatusBar.system.statusItem(withLength: NSStatusItem.variableLength)
        if let button = ramItem?.button {
            button.title = "🧠 --%"
            button.font = NSFont.systemFont(ofSize: 11)
        }

        // 每 2 秒刷新
        refreshTask = Task { [weak self] in
            while !Task.isCancelled {
                await self?.refresh()
                try? await Task.sleep(nanoseconds: 2_000_000_000)
            }
        }
    }

    func stop() {
        refreshTask?.cancel()
        if let item = tempItem { NSStatusBar.system.removeStatusItem(item) }
        if let item = cpuItem { NSStatusBar.system.removeStatusItem(item) }
        if let item = ramItem { NSStatusBar.system.removeStatusItem(item) }
    }

    private func refresh() async {
        let stats = await monitor.snapshot()
        await MainActor.run {
            cpuUsage = stats.cpuUsage
            ramUsage = stats.memoryUsagePercent
            temperature = stats.cpuTemperature
            memoryUsed = stats.memoryUsed
            memoryTotal = stats.memoryTotal

            tempItem?.button?.title = tempString(stats.cpuTemperature)
            cpuItem?.button?.title = cpuString(stats.cpuUsage)
            ramItem?.button?.title = ramString(stats.memoryUsagePercent)
        }
    }

    private func tempString(_ t: Double) -> String {
        if t <= 0 { return "🌡️ --°C" }
        return String(format: "🌡️ %.0f°C", t)
    }

    private func cpuString(_ pct: Double) -> String {
        String(format: "⚡ %.0f%%", pct * 100)
    }

    private func ramString(_ pct: Double) -> String {
        String(format: "🧠 %.0f%%", pct * 100)
    }
```

SwiftUI App 不支持 `init()` 直接写。使用 `.task` 或 `NSApplicationDelegateAdaptor`:

```swift
@main
struct MacCleanerApp: App {

    @StateObject private var appState = AppState()

    var body: some Scene {
        WindowGroup {
            ContentView()
                .environmentObject(appState)
                // ...
        }
    }
}

final class AppDelegate: NSObject, NSApplicationDelegate {
    func applicationDidFinishLaunching(_ notification: Notification) {
        MenuBarController.shared.start()
    }

    func applicationWillTerminate(_ notification: Notification) {
        MenuBarController.shared.stop()
    }
}
```
