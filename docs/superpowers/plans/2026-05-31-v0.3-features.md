/// 灵动岛菜单栏控制器 — 混合模式：常驻胶囊 + 事件弹窗
/// 替代旧版 NSStatusBar MenuBarController
@MainActor
final class DynamicIslandController: ObservableObject {
    static let shared = DynamicIslandController()

    private let monitor = SystemMonitor()
    private var timer: Timer?
    private var currentNotch: DynamicNotch<NotchHUDView>?
    private var highCPUStartTime: Date?
    private var highCPUNotified = false

    private init() {}

    // MARK: - Start / Stop

    func start() {
        // 显示常驻 HUD
        let hud = NotchHUDView(controller: self)
        let notch = DynamicNotch(content: hud)
        notch.show()
        currentNotch = notch

        // 初刷新
        Task { await refresh() }

        // 每 2 秒刷新
        timer = Timer.scheduledTimer(withTimeInterval: 2.0, repeats: true) { [weak self] _ in
            Task { @MainActor in
                await self?.refresh()
            }
        }
    }

    func stop() {
        timer?.invalidate()
        timer = nil
        currentNotch?.hide()
        currentNotch = nil
    }

    // MARK: - Refresh

    private func refresh() async {
        let stats = await monitor.snapshot()
        let cpuPct = Int(stats.cpuUsage * 100)
        let ramPct = Int(stats.memoryUsagePercent * 100)

        // 更新常驻 HUD 数据（通过 Combine 或直接重建）
        NotchHUDData.shared.cpuPercent = cpuPct
        NotchHUDData.shared.ramPercent = ramPct

        // CPU > 80% 持续 10s → 警告弹窗
        if cpuPct > 80 {
            if highCPUStartTime == nil {
                highCPUStartTime = Date()
            } else if !highCPUNotified,
                      let start = highCPUStartTime,
                      Date().timeIntervalSince(start) > 10 {
                highCPUNotified = true
                showCPUWarning(cpuPercent: cpuPct)
            }
        } else {
            highCPUStartTime = nil
            highCPUNotified = false
        }
    }

    // MARK: - Event Notifications

    /// 扫描完成通知
    func showScanComplete(fileCount: Int, duration: TimeInterval) {
        let notch = DynamicNotchInfo(
            icon: "magnifyingglass",
            title: "扫描完成",
            description: "发现 \(fileCount) 个文件，耗时 \(String(format: "%.0f", duration)) 秒"
        )
        notch.show(for: 4)
    }

    /// 清理完成通知
    func showCleanComplete(freedSize: String) {
        let notch = DynamicNotchInfo(
            icon: "trash",
            title: "清理完成",
            description: "已释放 \(freedSize)，文件在回收站保留 7 天"
        )
        notch.show(for: 4)
    }

            title: "CPU 使用率偏高",
            description: "当前 CPU 占用 \(cpuPercent)%，建议检查后台进程"
        )
        notch.show(for: 5)
    }

import Foundation

// MARK: - Trash Item Model

/// 回收站条目
struct TrashItem: Identifiable, Codable, Sendable {
    let id: String          // UUID string
    let originalPath: String
    let trashPath: String
    let size: Int64
    let fileName: String
    let deletedAt: Date
    let expiresAt: Date

    var remainingDays: Int {
        let seconds = expiresAt.timeIntervalSinceNow
        return max(0, Int(seconds / 86400))
    }

    var formattedSize: String {
        ByteCountFormatter.string(fromByteCount: size, countStyle: .file)
    }

    var formattedExpiresAt: String {
        let formatter = DateFormatter()
        formatter.locale = Locale(identifier: "zh_CN")
        formatter.dateFormat = "MM-dd HH:mm"
        return formatter.string(from: expiresAt)
    }
}

// MARK: - Manifest

/// 回收站清单（持久化到 .trash-manifest.json）
private struct TrashManifest: Codable {
    var items: [TrashItem]
}

// MARK: - TrashManager Actor

/// 回收站管理器 — 文件移入/恢复/彻底删除/定时清理
actor TrashManager {
    private let fileManager = FileManager.default
    private let safetyManager = SafetyManager()

    /// 回收站根目录
    let recycleBinURL: URL

    /// 清单文件路径
    private var manifestURL: URL {
        recycleBinURL.appendingPathComponent(".trash-manifest.json")
    }

    /// 当前清单（内存缓存）
    private var manifest: TrashManifest

    /// 定时清理 Timer
    private var purgeTimer: Timer?

    // MARK: - Init

    init() {
        let docs = fileManager.urls(for: .documentDirectory, in: .userDomainMask).first!
        recycleBinURL = docs.appendingPathComponent("MacCleanerApp 回收站")
        manifest = TrashManifest(items: [])

        // 确保目录存在
        try? fileManager.createDirectory(at: recycleBinURL, withIntermediateDirectories: true)
    }

    // MARK: - Lifecycle

    private var isStarted = false

    func start() {
        guard !isStarted else { return }
        isStarted = true

        loadManifest()

        // 定时清理（每 12 小时）
        purgeTimer = Timer.scheduledTimer(withTimeInterval: 12 * 3600, repeats: true) { [weak self] _ in
            Task { await self?.purgeExpired() }
        }
        // 启动时立即检查一次过期条目
        Task { await self.purgeExpired() }
    }

    func stop() {
        purgeTimer?.invalidate()
        purgeTimer = nil
    }

    // MARK: - Move to Trash

    /// 将文件移入回收站
    /// - Returns: 成功移入的条目列表
    @discardableResult
    func moveToTrash(_ urls: [URL]) async throws -> [TrashItem] {
        var newItems: [TrashItem] = []
        let now = Date()
        let expireDate = Calendar.current.date(byAdding: .day, value: 7, to: now) ?? now

        for url in urls {
            // 白名单检查
            guard await !safetyManager.isProtected(url) else { continue }
            guard fileManager.fileExists(atPath: url.path) else { continue }

            let itemID = UUID().uuidString
            let itemDir = recycleBinURL.appendingPathComponent(itemID)
            let destURL = itemDir.appendingPathComponent(url.lastPathComponent)

            do {
                try fileManager.createDirectory(at: itemDir, withIntermediateDirectories: true)
                try fileManager.moveItem(at: url, to: destURL)

                let fileSize: Int64 = {
                    if let values = try? destURL.resourceValues(forKeys: [.fileSizeKey]),
                       let size = values.fileSize {
                        return Int64(size)
                    }
                    return 0
                }()

                let item = TrashItem(
                    id: itemID,
                    originalPath: url.path,
                    trashPath: destURL.path,
                    size: fileSize,
                    fileName: url.lastPathComponent,
                    deletedAt: now,
                    expiresAt: expireDate
                )
                newItems.append(item)
            } catch {
                // 如果 move 失败，跳过该文件
                try? fileManager.removeItem(at: itemDir)
                continue
            }
        }

        // 写入清单
        manifest.items.append(contentsOf: newItems)
        saveManifest()

        return newItems
    }

    // MARK: - Restore

    /// 将文件从回收站移回原路径
    func restore(_ item: TrashItem) async throws {
        let trashURL = URL(fileURLWithPath: item.trashPath)
        let originalURL = URL(fileURLWithPath: item.originalPath)

        guard fileManager.fileExists(atPath: trashURL.path) else {
            throw TrashError.fileNotFound
        }

        // 确保原路径的父目录存在
        try fileManager.createDirectory(
            at: originalURL.deletingLastPathComponent(),
            withIntermediateDirectories: true
        )

        try fileManager.moveItem(at: trashURL, to: originalURL)

        // 清理回收站中的空目录
        let itemDir = trashURL.deletingLastPathComponent()
        try? fileManager.removeItem(at: itemDir)

        // 从清单移除
        manifest.items.removeAll { $0.id == item.id }
        saveManifest()
    }

    // MARK: - Permanent Delete

    /// 彻底删除（移入系统废纸篓）
    func permanentlyDelete(_ item: TrashItem) async throws {
        let trashURL = URL(fileURLWithPath: item.trashPath)

        guard fileManager.fileExists(atPath: trashURL.path) else {
            // 文件已不存在，从清单移除
            manifest.items.removeAll { $0.id == item.id }
            saveManifest()
            return
        }

        // 移入系统废纸篓
        var resultingItemURL: NSURL?
        try fileManager.trashItem(at: trashURL, resultingItemURL: &resultingItemURL)

        // 清理回收站空目录
        let itemDir = trashURL.deletingLastPathComponent()
        try? fileManager.removeItem(at: itemDir)

        // 从清单移除
        manifest.items.removeAll { $0.id == item.id }
        saveManifest()
    }

    // MARK: - Query

    /// 获取所有回收站条目
    func getItems() -> [TrashItem] {
        manifest.items.sorted { $0.deletedAt > $1.deletedAt }
    }

    // MARK: - Purge

    /// 清理过期条目 → 移入系统废纸篓
    func purgeExpired() async {
        let now = Date()
        let expired = manifest.items.filter { $0.expiresAt < now }

        for item in expired {
            let trashURL = URL(fileURLWithPath: item.trashPath)
            if fileManager.fileExists(atPath: trashURL.path) {
                var resultingItemURL: NSURL?
                try? fileManager.trashItem(at: trashURL, resultingItemURL: &resultingItemURL)
            }
            // 清理空目录
            let itemDir = trashURL.deletingLastPathComponent()
            try? fileManager.removeItem(at: itemDir)
        }

        if !expired.isEmpty {
            manifest.items.removeAll { $0.expiresAt < now }
            saveManifest()
        }
    }

    // MARK: - Manifest I/O

    private func loadManifest() {
        guard let data = try? Data(contentsOf: manifestURL),
              let decoded = try? JSONDecoder().decode(TrashManifest.self, from: data) else {
            manifest = TrashManifest(items: [])
            return
        }
        manifest = decoded
    }

    private func saveManifest() {
        guard let data = try? JSONEncoder().encode(manifest) else { return }
        try? data.write(to: manifestURL, options: .atomic)
    }
}

// MARK: - Errors

enum TrashError: LocalizedError {
    case fileNotFound

    var errorDescription: String? {
        switch self {
        case .fileNotFound: return "文件在回收站中不存在"
        }
    }
}
